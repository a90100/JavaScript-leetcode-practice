# 904. Fruit Into Baskets

## 解題程式碼

第一次解題：

```javascript
var totalFruit = function (fruits) {
  let max = 0;
  let l = 0;
  const fruitsMap = new Map();

  for (let i = 0; i < fruits.length; i++) {
    if (!fruitsMap.has(fruits[i]) && fruitsMap.size > 1) {
      const [fruitsType1, fruitsType2] = [...fruitsMap.keys()];
      while (fruitsMap.get(fruitsType1) > 0 && fruitsMap.get(fruitsType2) > 0) {
        fruitsMap.set(fruits[l], fruitsMap.get(fruits[l++]) - 1);
      }
      if (fruitsMap.get(fruitsType1) === 0) fruitsMap.delete(fruitsType1);
      if (fruitsMap.get(fruitsType2) === 0) fruitsMap.delete(fruitsType2);
    }

    fruitsMap.set(fruits[i], (fruitsMap.get(fruits[i]) ?? 0) + 1);
    max = Math.max(max, i - l + 1);
  }

  return max;
};
```

更精簡：

```javascript
var totalFruit = function (fruits) {
  let max = 0;
  let l = 0;
  const fruitsMap = new Map();

  for (let i = 0; i < fruits.length; i++) {
    fruitsMap.set(fruits[i], (fruitsMap.get(fruits[i]) ?? 0) + 1);

    while (fruitsMap.size > 2) {
      fruitsMap.set(fruits[l], fruitsMap.get(fruits[l]) - 1);
      if (fruitsMap.get(fruits[l]) === 0) fruitsMap.delete(fruits[l]);
      l++;
    }

    max = Math.max(max, i - l + 1);
  }

  return max;
};
```
