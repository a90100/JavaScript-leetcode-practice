# 384. Shuffle an Array

## 解題程式碼

解法 1.

```javascript
var Solution = function (nums) {
  this.nums = nums;
};

Solution.prototype.reset = function () {
  return this.nums;
};

Solution.prototype.shuffle = function () {
  let index = this.nums.length - 1;
  const nums = [...this.nums];

  while (index > 0) {
    const randomIndex = Math.floor(Math.random() * (index + 1));
    [nums[index], nums[randomIndex]] = [nums[randomIndex], nums[index]];

    index--;
  }
  return nums;
};
```

解法 2.

```javascript
var Solution = function (nums) {
  this.nums = nums;
};

Solution.prototype.reset = function () {
  return this.nums;
};

Solution.prototype.shuffle = function () {
  const nums = [...this.nums];
  for (let i = 0; i < nums.length; i++) {
    let changeIndex = i + Math.floor(Math.random() * (nums.length - i));

    [nums[i], nums[changeIndex]] = [nums[changeIndex], nums[i]];
  }
  return nums;
};
```
