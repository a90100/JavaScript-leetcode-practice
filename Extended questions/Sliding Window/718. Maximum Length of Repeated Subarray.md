# 718. Maximum Length of Repeated Subarray

## 解題程式碼

動態規劃解法：

```javascript
var findLength = function (nums1, nums2) {
  let max = 0;
  const DP = new Array(nums1.length + 1).fill().map(() => new Array(nums2.length + 1).fill(0));

  for (let i = 1; i < nums1.length + 1; i++) {
    for (let j = 1; j < nums2.length + 1; j++) {
      if (nums1[i - 1] === nums2[j - 1]) {
        DP[i][j] = DP[i - 1][j - 1] + 1;
      }
      max = Math.max(max, DP[i][j]);
    }
  }

  return max;
};
```

Sliding window 解法：

```javascript
var findLength = function (nums1, nums2) {
  const findMaxLen = (arr1, arr2) => {
    let totalMax = 0;

    for (let i = 0; i < arr1.length; i++) {
      let curMax = 0;
      let temp = 0;

      for (let j = 0; j < arr2.length && i + j < arr1.length; j++) {
        if (arr1[i + j] === arr2[j]) {
          temp++;
        } else {
          temp = 0;
        }
        curMax = Math.max(curMax, temp);
      }
      totalMax = Math.max(curMax, totalMax);
    }

    return totalMax;
  };

  return Math.max(findMaxLen(nums1, nums2), findMaxLen(nums2, nums1));
};
```

第一次 Submit 時出了錯，發現多加上第二組遍歷就可以通過...

```javascript
var findLength = function (nums1, nums2) {
  let max = 0;

  // 遍歷 nums1 一次，維護兩個指針，指向 nums2 相同子陣列的起始和結束位置
  for (let i = 0; i < nums1.length; i++) {
    let nums2LeftP = 0;
    let nums2RightP = 0;

    while (i + nums2RightP < nums1.length && nums2RightP < nums2.length) {
      if (nums1[i + nums2RightP] !== nums2[nums2RightP]) nums2LeftP = nums2RightP + 1; // 把 sliding window 範圍重置
      nums2RightP++;
      max = Math.max(max, nums2RightP - nums2LeftP);
    }
  }

  // 舉例
  // 第一次比對：i = 0
  // nums1 = [1,2,3,2,1]
  // nums2 = [3,2,1,4,7]
  // 第二次比對：i = 1
  // nums1 =   [1,2,3,2,1]
  // nums2 = [3,2,1,4,7]
  // 第三次比對：i = 2
  // nums1 =     [1,2,3,2,1]
  // nums2 = [3,2,1,4,7]
  // ...以此類推

  // 加上了以下這段後通過所有測資
  for (let i = 0; i < nums2.length; i++) {
    let nums1LeftP = 0;
    let nums1RightP = 0;

    while (i + nums1RightP < nums2.length && nums1RightP < nums1.length) {
      if (nums2[i + nums1RightP] !== nums1[nums1RightP]) nums1LeftP = nums1RightP + 1;
      nums1RightP++;
      max = Math.max(max, nums1RightP - nums1LeftP);
    }
  }

  // 舉例
  // 第一次比對：i = 0
  // nums1 = [1,2,3,2,1]
  // nums2 = [3,2,1,4,7]
  // 第二次比對：i = 1
  // nums1 = [1,2,3,2,1]
  // nums2 =   [3,2,1,4,7]
  // 第三次比對：i = 2
  // nums1 = [1,2,3,2,1]
  // nums2 =     [3,2,1,4,7]
  // ...以此類推

  return max;
};
```
