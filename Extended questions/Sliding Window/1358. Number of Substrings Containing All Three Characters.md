# 1358. Number of Substrings Containing All Three Characters

## 解題程式碼

第一次解題：

```javascript
var numberOfSubstrings = function (s) {
  let a = 0;
  let b = 0;
  let c = 0;
  let l = 0;
  let res = 0;

  for (let i = 0; i < s.length; i++) {
    if (s[i] === 'a') a++;
    if (s[i] === 'b') b++;
    if (s[i] === 'c') c++;
    if (a > 0 && b > 0 && c > 0) res += s.length - i;

    while (a > 0 && b > 0 && c > 0) {
      if (s[l] === 'a') a--;
      if (s[l] === 'b') b--;
      if (s[l] === 'c') c--;

      l++;
      if (a > 0 && b > 0 && c > 0) res += s.length - i;
    }
  }

  return res;
};
```

化簡：

```javascript
var numberOfSubstrings = function (s) {
  const charMap = { a: 0, b: 0, c: 0 };
  let l = 0;
  let res = 0;

  for (let i = 0; i < s.length; i++) {
    charMap[s[i]]++;

    while (charMap.a > 0 && charMap.b > 0 && charMap.c > 0) {
      res += s.length - i;
      charMap[s[l++]]--;
    }
  }

  return res;
};
```
