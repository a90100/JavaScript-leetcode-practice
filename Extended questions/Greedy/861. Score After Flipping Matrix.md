# 861. Score After Flipping Matrix

## 解題程式碼

```javascript
var matrixScore = function (grid) {
  // 最好還是 1 開頭，0111 => 7，1000 => 8
  // 記錄同一直欄出現 0 的次數
  const columnZeroNum = new Array(grid[0].length).fill(0);
  for (let i = 0; i < grid.length; i++) {
    let isRevertRow = grid[i][0] === 0;
    for (let j = 0; j < grid[i].length; j++) {
      if (isRevertRow) grid[i][j] = grid[i][j] === 1 ? 0 : 1;
      if (grid[i][j] === 0) columnZeroNum[j]++;
    }
  }

  // 翻轉較多 0 的直欄
  for (let i = 0; i < columnZeroNum.length; i++) {
    if (columnZeroNum[i] > Math.floor(grid.length / 2)) {
      for (let j = 0; j < grid.length; j++) {
        grid[j][i] = grid[j][i] === 1 ? 0 : 1;
      }
    }
  }

  return grid.reduce((tol, cur) => tol + +parseInt(cur.join(''), 2), 0);
};
```
