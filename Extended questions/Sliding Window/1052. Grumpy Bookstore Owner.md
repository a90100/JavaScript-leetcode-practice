# 1052. Grumpy Bookstore Owner

## 解題程式碼

```javascript
var maxSatisfied = function (customers, grumpy, minutes) {
  let satisfied = 0;
  let maxBounsSatisfied = 0;
  let curBounsSatisfied = 0;

  for (let i = 0; i < customers.length; i++) {
    if (grumpy[i] === 0) satisfied += customers[i]; // 把不開技的滿意人數加起來
    if (grumpy[i] === 1) curBounsSatisfied += customers[i];
    if (i >= minutes && grumpy[i - minutes] === 1) curBounsSatisfied -= customers[i - minutes];
    maxBounsSatisfied = Math.max(maxBounsSatisfied, curBounsSatisfied);
  }

  return satisfied + maxBounsSatisfied;
};
```
